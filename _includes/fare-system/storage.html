Minimum fare per cell / airline / product

SELECT route, departure, stops, cabin, trip,
       stay_nights, country_of_sale, carrier,
       product_group, MIN(total_usd_minor) AS fare
FROM offers
WHERE customer = :customer AND reporting_day = :day
  AND collection_run = :run
GROUP BY 1,2,3,4,5,6,7,8,9;
